def replacement_data(node,index):
    if isinstance(node,ast.Constant):
        return node.value
    if isinstance(node,ast.Name):
        assert node.id=='n'
        return index
    if isinstance(node,ast.List):
        return [replacement_data(x,index) for x in node.elts]
    if isinstance(node,ast.Tuple):
        return tuple(replacement_data(x,index) for x in node.elts)
    if isinstance(node,ast.Dict):
        keys=[replacement_data(x,index) for x in node.keys]
        assert len(keys)==len(set(keys))
        return dict(zip(keys,[replacement_data(x,index) for x in node.values],strict=True))
    if isinstance(node,ast.JoinedStr):
        return ''.join(replacement_data(x,index) for x in node.values)
    if isinstance(node,ast.FormattedValue):
        assert node.conversion==-1 and node.format_spec is None
        return str(replacement_data(node.value,index))
    if isinstance(node,ast.BinOp):
        assert isinstance(node.op,ast.Add)
        return replacement_data(node.left,index)+replacement_data(node.right,index)
    if isinstance(node,ast.Call):
        assert isinstance(node.func,ast.Name) and node.func.id=='str'
        assert len(node.args)==1 and not node.keywords
        return str(replacement_data(node.args[0],index))
    raise AssertionError(('forbidden_replacement_data',ast.dump(node)))
def gated_bytes(path):
    path=Path(path).resolve()
    key=str(path.relative_to(Path.cwd()))
    payload=json.loads((OUT.parent/'source-payload.json').read_text())
    expected=payload['sources'][key].encode('utf-8','strict')
    actual=path.read_bytes()
    assert actual==expected, ('gated_full_bytes_changed',key)
    gate=json.loads((OUT.parent/'compile-gate.json').read_text())
    entry=gate['sources'][key]
    assert entry=={'bytes':len(actual),'sha256':hashlib.sha256(actual).hexdigest()}
    assert gate['options']=={'mode':'exec','flags':0,'dont_inherit':True,'optimize':0}
    return actual

def part(source, first, last):
    assert source.count(first)==1 and source.count(last)==1
    assert source.index(first)<source.index(last)
    return source[source.index(first):source.index(last)]
def once(source,old,new,records):
    assert source.count(old)==1, ('unique_fresh_adaptation',old)
    records.append({'old':old,'new':new,'occurrences':1})
    return source.replace(old,new,1)

    save('audit-manifest.json',{'python':sys.version,'indices':list(INDICES),'time_cap_seconds':180,'byte_cap':LIMIT,
       'start_before_imports_reads_and_setup':True,'compile_before_any_child_or_generated_execution':True,
       'old_entrypoints_executed':False,'producer_executed':False,'no_retry_or_resumption':True,
       'signed_checks':'exact lexical installer; four inert preflights before sole replay per index',
       'historical_timing_qualifications_retained':True})
    stage='four_inert_preflights'
    ns={}
    assert (OUT/'preflight.py').read_bytes()==executable[str((OUT/'preflight.py').relative_to(ROOT))].encode()
    exec(codes[str((OUT/'preflight.py').relative_to(ROOT))],ns)
    assert (OUT/'installer.py').read_bytes()==installer.encode()
    result=ns['preflight'](OUT,installer)
    save('identity-preflight.json',{'status':'PASS','indices':list(INDICES),'results':result,
       'all_before_any_kernel':True,'elapsed_seconds':time.monotonic()-START})
    workcheck()
    stage='complete_numerical_audits'
    for index in INDICES:
        current=index
        workcheck()
        assert inventory([ROOT/p for p in sources])==sources
        for rel,text in executable.items():
            assert (ROOT/rel).read_bytes()==text.encode('utf-8','strict')
        dest=OUT/('n-'+str(index))
        dest.mkdir()
        child_start=time.monotonic()-START
        try:
            child=subprocess.run([sys.executable,'-B',str(OUT/'worker.py'),str(index),str(dest),repr(START)],
                 capture_output=True,timeout=max(.001,174-(time.monotonic()-START)))
            stdout,stderr,code=child.stdout,child.stderr,child.returncode
        except subprocess.TimeoutExpired as exc:
            stdout,stderr,code=exc.stdout or b'',exc.stderr or b'',None
            failure={'type':'ChildTimeout','n':index,'message':'killed and waited; no retry'}
        assert len(stdout)+len(stderr)<=8192
        rawsave(dest/'child-stdout.txt',stdout)
        rawsave(dest/'child-stderr.txt',stderr)
        exits.append({'n':index,'returncode':code,'start_seconds':child_start,'observed_exit_seconds':time.monotonic()-START})
        save('observed-child-exits.json',exits)
        if failure is not None:
            break
        if code!=0:
            failure={'type':'ChildFailure','n':index,'returncode':code,'evidence':str((dest/'first-failure.json').relative_to(ROOT))}
            break
        status=load(dest/'status.json')
        assert status['status']=='PASS' and status['state']=={'n':index,'stage':'complete','panels':1016} and stderr==b''
        assert load(dest/'child-stdout.txt')=={'n':index,'status':'PASS','panels':1016,'coefficient_pairs':21264880,'strict_radius_powers':[183,50]}
        completed.append(index)
        save('completed-indices.json',completed)
        print(json.dumps({'n':index,'status':'PASS','observed_exit_seconds':exits[-1]['observed_exit_seconds']}),flush=True)
        workcheck()
except BaseException as exc:
    if failure is None:
        failure={'type':type(exc).__name__,'message':str(exc),'n':current,'stage':stage,'traceback':traceback.format_exc()}
finally:
    signal.setitimer(signal.ITIMER_REAL,max(.001,180-(time.monotonic()-START)))
    final=inventory([ROOT/p for p in initial])
    mismatches=[p for p in initial if initial[p]!=final.get(p)]
    fresh_ok=inventory([ROOT/p for p in sources])==sources
    all_emitted_ok=all((ROOT/rel).read_bytes()==text.encode('utf-8','strict') for rel,text in executable.items())
    preserved=not mismatches and tree_names()==names and fresh_ok and all_emitted_ok
    save('preservation.json',{'count':len(initial),'files':final,'mismatches':mismatches,
        'all_match':preserved,'original_tree_file_sets_equal':tree_names()==names,
        'fresh_sources_equal':fresh_ok,'all_executable_bytes_equal':all_emitted_ok})
    if not preserved and failure is None:
        failure={'type':'PreservationFailure','message':str(mismatches)}
    if failure is not None:
        save('first-failure.json',failure)
    elapsed=time.monotonic()-START
    sec=math.ceil(elapsed)+2
    size=disk()+RESERVE
    okay=failure is None and completed==list(INDICES) and sec<=180 and size<=LIMIT
    terminal={'status':'PASS' if okay else 'STOP','completed_indices':completed,'failure':failure,
       'child_exits':exits,'preserved_inputs':len(initial),'preservation':preserved,
       'elapsed_before_terminal_seconds':elapsed,'inclusive_seconds':sec,'inclusive_new_bytes':size,
       'required_seconds':180,'required_new_bytes':LIMIT,'terminal_timer_retained':True,
       'path':str((OUT/'terminal-report.json').relative_to(ROOT))}
    save('terminal-report.json',terminal)
    assert inventory([ROOT/p for p in initial])==initial and tree_names()==names
    assert inventory([ROOT/p for p in sources])==sources
    assert all((ROOT/rel).read_bytes()==text.encode('utf-8','strict') for rel,text in executable.items())
    save('finalization.json',{'elapsed_after_terminal_seconds':time.monotonic()-START,
       'new_bytes_after_terminal':disk(),'inclusive_seconds':sec,'inclusive_new_bytes':size,
       'all_children_observed_before_terminal':True,'preservation_rechecked_after_terminal':True,
       'parent_exit_time_not_claimed':True,'terminal_timer_retained':True})
    assert time.monotonic()-START<=sec<=180 and disk()+8192<=size<=LIMIT
    print(json.dumps(terminal),flush=True)
    assert time.monotonic()-START<=sec
    sys.exit(0 if okay else 1)
